PASO 4: Definir Estado del Agente # ============================================ class AgentState(TypedDict): messages: Annotated[Sequence[BaseMessage], operator.add] next: str print("✅ Estado definido") # ============================================ # PASO 5: Crear Herramientas # ============================================ def calculadora(expression: str) -> str: """Herramienta simple de calculadora""" try: result = eval(expression) return f"Resultado: {result}" except Exception as e: return f"Error al calcular: {str(e)}" def buscar_info(query: str) -> str: """Simula una búsqueda de información""" info_base = { "python": "Python es un lenguaje de programación de alto nivel, interpretado y de propósito general.", "ia": "La Inteligencia Artificial es la simulación de procesos de inteligencia humana por máquinas.", "langgraph": "LangGraph es una librería para construir agentes con estados utilizando grafos.", "qwen": "Qwen es una serie de modelos de lenguaje desarrollados por Alibaba Cloud.", } query_lower = query.lower() for key, value in info_base.items(): if key in query_lower: return value return "No encontré información específica sobre esa consulta." print("✅ Herramientas creadas") # ============================================ # PASO 6: Definir Nodos del Agente # ============================================ def agente_razonamiento(state: AgentState): """Nodo que procesa la consulta del usuario""" messages = state["messages"] ultima_mensaje = messages[-1].content # Determinar si necesita usar herramientas if any(word in ultima_mensaje.lower() for word in ["calcular", "calcula", "suma", "resta", "multiplica", "divide"]): return {"next": "calculadora"} elif any(word in ultima_mensaje.lower() for word in ["busca", "qué es", "información", "explica"]): return {"next": "buscar"} else: return {"next": "responder"} def nodo_calculadora(state: AgentState): """Nodo que usa la calculadora""" messages = state["messages"] ultima_mensaje = messages[-1].content import re expresion = re.search(r'[\d\+\-\*/\(\)\s]+', ultima_mensaje) if expresion: resultado = calculadora(expresion.group()) else: resultado = "No pude identificar una expresión matemática válida." return { "messages": [AIMessage(content=f"🧮 {resultado}")], "next": "fin" } def nodo_buscar(state: AgentState): """Nodo que busca información""" messages = state["messages"] ultima_mensaje = messages[-1].content resultado = buscar_info(ultima_mensaje) return { "messages": [AIMessage(content=f"📚 {resultado}")], "next": "fin" } def nodo_responder(state: AgentState): """Genera respuesta con Qwen""" messages = state["messages"] ultimo_mensaje = messages[-1].content # Llamar a Qwen con la función helper response = llamar_qwen(ultimo_mensaje) return { "messages": [AIMessage(content=f"🤖 {response}")], "next": "fin" } # Generar respuesta response = llm.invoke(prompt) return { "messages": [AIMessage(content=f"🤖 {response}")], "next": "fin" } def deberia_continuar(state: AgentState): """Decide si continuar o terminar""" return state.get("next", "fin") print("✅ Nodos definidos") # ============================================ # PASO 7: Construir el Grafo # ============================================ workflow = StateGraph(AgentState) # Añadir nodos workflow.add_node("razonamiento", agente_razonamiento) workflow.add_node("calculadora", nodo_calculadora) workflow.add_node("buscar", nodo_buscar) workflow.add_node("responder", nodo_responder) # Punto de entrada workflow.set_entry_point("razonamiento") # Edges condicionales workflow.add_conditional_edges( "razonamiento", deberia_continuar, { "calculadora": "calculadora", "buscar": "buscar", "responder": "responder", "fin": END } ) # Edges finales workflow.add_edge("calculadora", END) workflow.add_edge("buscar", END) workflow.add_edge("responder", END) # Compilar app = workflow.compile() print("✅ Grafo construido y compilado") # ============================================ #
